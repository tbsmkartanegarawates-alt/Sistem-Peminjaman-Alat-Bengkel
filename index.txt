        // Render Catalog Cards
        function renderTools() {
            const grid = document.getElementById('toolsGrid');
            const searchVal = document.getElementById('searchTool').value.toLowerCase();
            
            grid.innerHTML = '';

            const filtered = toolCatalog.filter(tool => {
                const matchCat = activeCategory === 'All' || tool.kat === activeCategory;
                const matchSearch = tool.nama.toLowerCase().includes(searchVal);
                return matchCat && matchSearch;
            });

            if(filtered.length === 0) {
                grid.innerHTML = `
                    <div class="col-span-full py-8 text-center text-slate-400">
                        <i class="fa-solid fa-box-open text-3xl mb-2"></i>
                        <p class="text-xs">Alat tidak ditemukan.</p>
                    </div>`;
                return;
            }

            filtered.forEach(tool => {
                const qtySelected = selectedCart[tool.id] || 0;
                const isSelected = qtySelected > 0;

                const card = document.createElement('div');
                card.className = `p-3.5 rounded-xl border transition-all duration-200 flex flex-col justify-between ${
                    isSelected ? 'border-blue-500 bg-blue-50/40 ring-1 ring-blue-500 shadow-sm' : 'border-slate-200 bg-white hover:border-slate-300'
                }`;

                card.innerHTML = `
                    <div>
                        <div class="flex items-start justify-between gap-2 mb-1.5">
                            <div class="p-2 ${isSelected ? 'bg-blue-600 text-white' : 'bg-slate-100 text-slate-600'} rounded-lg text-sm transition">
                                <i class="fa-solid ${tool.icon}"></i>
                            </div>
                            <span class="text-[10px] font-bold px-2 py-0.5 rounded ${isSelected ? 'bg-blue-200 text-blue-800' : 'bg-slate-100 text-slate-500'}">
                                Stok: ${tool.stok} ${tool.satuan}
                            </span>
                        </div>
                        <h3 class="font-bold text-slate-800 text-xs sm:text-sm line-clamp-2">${tool.nama}</h3>

                    </div>

                    <div class="mt-3 pt-2 border-t border-slate-100 flex items-center justify-between">
                        <span class="text-xs font-semibold text-slate-600">Jumlah:</span>
                        <div class="flex items-center space-x-2">
                            <button type="button" onclick="updateQty('${tool.id}', -1)" 
                                class="w-7 h-7 bg-slate-200 hover:bg-slate-300 rounded-md font-bold text-slate-700 text-xs flex items-center justify-center transition disabled:opacity-40"
                                ${qtySelected === 0 ? 'disabled' : ''}>-</button>
                            <span class="text-xs font-extrabold w-5 text-center text-blue-900">${qtySelected}</span>
                            <button type="button" onclick="updateQty('${tool.id}', 1)" 
                                class="w-7 h-7 bg-blue-600 hover:bg-blue-700 text-white rounded-md font-bold text-xs flex items-center justify-center transition disabled:opacity-40"

                        </div>
                    </div>
                `;
                grid.appendChild(card);
            });
        }

        // Update Cart Item Quantity
        function updateQty(toolId, delta) {
            const tool = toolCatalog.find(t => t.id === toolId);
            if(!tool) return;

            const current = selectedCart[toolId] || 0;
            const updated = current + delta;

            if(updated <= 0) {
                delete selectedCart[toolId];
            } else if(updated <= tool.stok) {
                selectedCart[toolId] = updated;
            } else {
                showToast(`Stok ${tool.nama} terbatas!`, 'warning');
                return;
            }

            renderTools();
            updateCartSummary();
        }

        // Render Cart Summary in Form
        function updateCartSummary() {
            const summaryContainer = document.getElementById('cartSummaryList');
            const badge = document.getElementById('cartCountBadge');
            
            const keys = Object.keys(selectedCart);
            let totalItemCount = 0;

            if(keys.length === 0) {
                summaryContainer.innerHTML = `<p class="text-slate-400 italic">Belum ada alat yang dipilih dari daftar di sebelah kanan.</p>`;
                badge.innerText = '0 Item';
                return;
            }

            summaryContainer.innerHTML = '';
            keys.forEach(id => {
                const tool = toolCatalog.find(t => t.id === id);
                const qty = selectedCart[id];
                totalItemCount += qty;

                const row = document.createElement('div');
                row.className = 'flex items-center justify-between bg-white px-2 py-1 rounded border border-slate-200';
                row.innerHTML = `
                    <span class="font-medium truncate text-slate-700 max-w-[180px]">${tool.nama}</span>
                    <span class="font-bold text-blue-700 bg-blue-50 px-1.5 py-0.5 rounded">${qty} ${tool.satuan}</span>
                `;
                summaryContainer.appendChild(row);
            });

            badge.innerText = `${totalItemCount} Total Alat`;
        }

        // Handle Loan Submission
        function handleFormSubmit(e) {
            e.preventDefault();

            const namaSiswa = document.getElementById('namaSiswa').value.trim();
            const kelas = document.getElementById('kelas').value.trim();
            const namaMapel = document.getElementById('namaMapel').value.trim();
            const namaGuru = document.getElementById('namaGuru').value.trim();

            const cartKeys = Object.keys(selectedCart);

            if(cartKeys.length === 0) {
                showToast('Harap pilih minimal 1 alat yang akan dipinjam!', 'warning');
                return;
            }

            // Build item list object
            const items = cartKeys.map(id => {
                const tool = toolCatalog.find(t => t.id === id);
                return {
                    id: tool.id,
                    nama: tool.nama,
                    jumlah: selectedCart[id],
                    satuan: tool.satuan
                };
            });

            const newId = 'PINJAM-' + Math.floor(1000 + Math.random() * 9000);
            const now = new Date().toLocaleString('id-ID');

            const loanRecord = {
                id: newId,
                waktu: now,
                siswa: namaSiswa,
                kelas: kelas,
                mapel: namaMapel,
                guru: namaGuru,
                items: items,
                status: 'Dipinjam',
                waktuKembali: '-'
            };

            // Add to Local Array
            loansData.unshift(loanRecord);
            updateBadgeCount();

            // Check if Google Apps Script is integrated
            if (typeof google !== 'undefined' && google.script && google.script.run) {
                google.script.run.withSuccessHandler(res => {
                    console.log('Saved to GAS Sheet:', res);
                }).simpanPeminjaman(loanRecord);
            }

            // Show Receipt Modal
            showReceipt(loanRecord);

            // Reset Form & Cart
            document.getElementById('borrowForm').reset();
            selectedCart = {};
            renderTools();
            updateCartSummary();

            showToast('Peminjaman berhasil dicatat!', 'success');
        }

        // Render Active Loans Table
        function renderLoansTable() {
            const tbody = document.getElementById('loansTableBody');
            const searchVal = document.getElementById('searchLoan').value.toLowerCase();

            tbody.innerHTML = '';

            const filtered = loansData.filter(l => 
                l.siswa.toLowerCase().includes(searchVal) || 
                l.kelas.toLowerCase().includes(searchVal) ||
                l.id.toLowerCase().includes(searchVal)
            );

            if(filtered.length === 0) {
                tbody.innerHTML = `
                    <tr>
                        <td colspan="6" class="p-8 text-center text-slate-400">
                            <i class="fa-solid fa-clipboard-check text-3xl mb-2"></i>
                            <p>Tidak ada data peminjaman ditemukan.</p>
                        </td>
                    </tr>`;
                return;
            }

            filtered.forEach(loan => {
                const itemListHTML = loan.items.map(i => 
                    `<span class="inline-block bg-slate-100 text-slate-700 px-2 py-0.5 rounded text-[11px] mr-1 mb-1 border border-slate-200">
                        ${i.nama} <b>(${i.jumlah} ${i.satuan})</b>
                     </span>`

                        <div class="text-[11px] text-slate-400">${loan.waktu}</div>
                    </td>
                    <td class="p-3">
                        <div class="font-bold text-slate-800">${loan.siswa}</div>
                        <div class="text-xs text-slate-500">${loan.kelas}</div>
                    </td>
                    <td class="p-3">
                        <div class="font-medium text-slate-700">${loan.mapel}</div>
                        <div class="text-xs text-slate-400">Guru: ${loan.guru}</div>
                    </td>
                    <td class="p-3 max-w-xs">
                        <div class="flex flex-wrap">${itemListHTML}</div>
                    </td>
                    <td class="p-3">
                        ${isReturned 
                            ? `<span class="bg-emerald-100 text-emerald-800 font-bold px-2.5 py-1 rounded-full text-[11px] inline-flex items-center gap-1"><i class="fa-solid fa-check"></i> Dikembalikan</span>` 
                            : `<span class="bg-amber-100 text-amber-800 font-bold px-2.5 py-1 rounded-full text-[11px] inline-flex items-center gap-1 animate-pulse"><i class="fa-solid fa-clock"></i> Dipinjam</span>`}
                    </td>
                    <td class="p-3 text-center">
                        ${!isReturned ? `
                            <button onclick="markAsReturned('${loan.id}')" class="px-3 py-1.5 bg-emerald-600 hover:bg-emerald-700 text-white font-bold rounded-lg text-xs shadow transition flex items-center gap-1 mx-auto">
                                <i class="fa-solid fa-arrow-rotate-left"></i> Kembalikan
                            </button>
                        ` : `
                            <span class="text-[11px] text-slate-400">${loan.waktuKembali}</span>
                        `}
                    </td>
                `;
                tbody.appendChild(tr);
            });
        }

        // Action: Mark loan as returned
        function markAsReturned(id) {
            const loan = loansData.find(l => l.id === id);
            if(loan) {
                loan.status = 'Dikembalikan';
                loan.waktuKembali = new Date().toLocaleString('id-ID');

                // Update in Google Apps Script if connected
                if (typeof google !== 'undefined' && google.script && google.script.run) {
                    google.script.run.updateStatusKembali(id);
                }

                renderLoansTable();
                updateBadgeCount();
                showToast(`Peminjaman ${id} selesai dikembalikan!`, 'success');
            }
        }

        function updateBadgeCount() {
            const activeCount = loansData.filter(l => l.status === 'Dipinjam').length;
            document.getElementById('badge-active-count').innerText = activeCount;
        }

// Simpan Data Peminjaman Baru
function simpanPeminjaman(data) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName('Peminjaman');
    
    if (!sheet) {
      sheet = ss.insertSheet('Peminjaman');
      sheet.appendRow([
        'ID Pinjam', 'Tanggal Pinjam', 'Nama Siswa', 'Kelas', 
        'Nama Mapel', 'Guru Pengampu', 'Daftar Alat', 'Status', 'Tanggal Kembali'
      ]);
      sheet.getRange(1, 1, 1, 9).setFontWeight('bold').setBackground('#E2E8F0');
    }

    // Format daftar barang
    var listAlatStr = data.items.map(function(item) {
      return item.nama + ' (' + item.jumlah + ' ' + item.satuan + ')';
    }).join(', ');

    sheet.appendRow([
      data.id,
      data.waktu,
      data.siswa,
      data.kelas,
      data.mapel,
      data.guru,
      listAlatStr,
      data.status,
      data.waktuKembali || '-'
    ]);

    return { success: true, message: 'Data berhasil disimpan ke Spreadsheet' };
  } catch (err) {
    return { success: false, error: err.toString() };
  }
}

// Update Status Pengembalian Alat
function updateStatusKembali(idPinjam) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName('Peminjaman');
    if (!sheet) return { success: false, message: 'Sheet tidak ditemukan' };

    var data = sheet.getDataRange().getValues();
    for (var i = 1; i < data.length; i++) {
      if (data[i][0] === idPinjam) {
        sheet.getRange(i + 1, 8).setValue('Dikembalikan');
        sheet.getRange(i + 1, 9).setValue(new Date().toLocaleString('id-ID'));
        return { success: true };
      }
    }
    return { success: false, message: 'ID tidak ditemukan' };
  } catch (err) {
    return { success: false, error: err.toString() };
  }
}`;
